cmake_minimum_required(VERSION 3.22.1)
project(astor_local LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized inference" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ASTOR_BASELINE_GREEDY "Comparison test only" OFF)
if(ANDROID AND ASTOR_BASELINE_GREEDY)
    message(FATAL_ERROR "Baseline greedy mode is reserved for host comparisons")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp" "${CMAKE_BINARY_DIR}/llama")
add_library(astor_local SHARED astor_local.cpp)
if(ASTOR_BASELINE_GREEDY)
    target_compile_definitions(astor_local PRIVATE ASTOR_BASELINE_GREEDY)
endif()
if(ANDROID)
    target_link_libraries(astor_local PRIVATE llama android log)
else()
    target_compile_definitions(astor_local PRIVATE ASTOR_EVAL_SEED=42)
    target_include_directories(astor_local PRIVATE "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/linux")
    target_link_libraries(astor_local PRIVATE llama)
    set_target_properties(astor_local PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()
